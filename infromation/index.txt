import React, { useEffect, useRef } from "react";
import { View, Text, Image , Animated } from "react-native";
import { StatusBar } from "expo-status-bar";
import { SafeAreaView } from "react-native-safe-area-context";
import { FontAwesome5 } from "@expo/vector-icons";
import icon from "@/assets/images/Logo.png";
import { useNavigation } from "@react-navigation/native";
import { router } from "expo-router";



export default function SplashScreen() {

    const fadeAnimation = useRef(new Animated.Value(0)).current;
  const navigation = useNavigation();

  useEffect(() => {
    const timeoutId = setTimeout(() => {
      router.push('/splashScreen1')
    }, 3000);

    Animated.timing(fadeAnimation, {
      toValue: 1,
      duration: 1000,
      useNativeDriver: true,
    }).start();

    return () => clearTimeout(timeoutId); 
  }, [fadeAnimation, navigation]);  
  return (
    <SafeAreaView className="flex flex-1 bg-[#6434C4]">
      <StatusBar style="light" />
      <View className="flex-1 justify-center items-center bg-[#6434C4]">
        {/* <FontAwesome5 name="plane" size={60} color="white" /> */}
        <Image className="w-20 h-20" source={icon}  />
        <Text className="text-white font-bold text-4xl mt-4">
          EthioSky Academy
        </Text>
        <Text className="text-gray-100 text-center mt-2 text-xl font-medium">
          Ethiopian Aviation Entrance Preparation
        </Text>
      </View>
    </SafeAreaView>
  );
}
